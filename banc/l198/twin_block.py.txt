def _edge(src_gi, dst_gi):
    m = emask_wA & (src_all == src_gi) & (dst_all == dst_gi)
    return dict(n=int(m.sum()), syn=int(w_syn[m].sum()),
                w_eff=float((vals_all[m] * np.where(vals_all[m] < 0, BETA, 1.0)).sum()))


def select_twin():
    cand = np.where(POOL_T[gi_wA])[0]               # local idx on the base
    d = _drv[cand]
    with np.errstate(divide="ignore", invalid="ignore"):
        score = np.where(d > 0, np.abs(np.log(d / D_REL)), np.inf)
        tie = np.where(OW_RAW[gi_wA[cand]] > 0,
                       np.abs(np.log(OW_RAW[gi_wA[cand]] / M_REL)), np.inf)
    o = np.lexsort((gi_wA[cand], tie, score))       # score, then tie, then gi
    rows = []
    for j in o[:6]:
        g = int(gi_wA[cand[j]])
        ci = cell_info(g)
        ci.update(loc_wA=int(cand[j]), x_mean=float(_xbar[cand[j]]),
                  x_mean_seeds=[float(v) for v in _cen["x_mean"][:, cand[j]]],
                  delivered_drive=float(d[j]), drive_ratio=float(d[j] / D_REL),
                  abs_log_drive_ratio=float(score[j]),
                  mass_ratio=float(OW_RAW[g] / M_REL),
                  abs_log_mass_ratio=float(tie[j]),
                  in_edges=int((emask_wA & (dst_all == g)).sum()),
                  direct_onto={pk: _direct_onto(g, pk) for pk in _pool_gi},
                  edge_from_carrier=_edge(GI_CARRIER, g), edge_to_carrier=_edge(g, GI_CARRIER),
                  edge_from_relay=_edge(GI_RELAY, g), edge_to_relay=_edge(g, GI_RELAY),
                  edge_to_74914=_edge(g, GI_DOWN))
        rows.append(ci)
    return dict(
        rule="argmin |log(drive_cand/drive_relay)| over POOL_T; tie-break smaller "
             "|log(mass_cand/mass_relay)| (mass = raw out|w| onto the k=24-base "
             "closure), then gi; drive = x_mean (stored 3-seed census of the k=24 "
             "base, the stored L175 census_k24 file) x |Wb| column abs-sum (PREREG l198)",
        pool="in k=24 base & super_class == relay's (ventral_nerve_cord_intrinsic) "
             "& not in any prior arm & not motor & not DNg100",
        census_seeds=[801, 901, 3561], pool_n=int(len(cand)),
        pool_n_positive_drive=int((d > 0).sum()),
        pool_n_active=int((_xbar[cand] > 0.05).sum()),
        prior_arm_cells_n=len(PRIOR),
        relay=dict(gi=GI_RELAY, delivered_drive=D_REL, out_w_raw=M_REL,
                   out_w_eff=float(OW_EFF[GI_RELAY]), x_mean=float(_xbar[RELAY_LOC]),
                   out_edges=int(OW_N[GI_RELAY]), eff_sign=float(EFF_SIGN[GI_RELAY])),
        pick=rows[0], runners_up=rows[1:6])
