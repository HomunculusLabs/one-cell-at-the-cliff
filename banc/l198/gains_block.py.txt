def post_seed(seed):
    """In-run anchors, all evaluated BEFORE a fresh seed is rolled.
    (a) at 901: cutwA (+ sc25) vs raw_l135 — C0A, L175/L187 form VERBATIM.
    (d) at 3545: cutwA (+ sc25) vs the stored L175 raws (L187 form VERBATIM).
    (e) at 3609: cutwA (+ sc25 / sc25F / rcut / rcutF) vs the stored L187 raws
        — the relay-cut arm itself must reproduce bit-for-bit.
    Any failure = VOID before a single fresh seed is rolled."""
    if seed == 901:
        z135c = np.load("l135/raw_l135.npz")
        pairs = [("cutwA", "cutwA")] + [(a, l) for a, l in (("sc25", "cutng25"),)
                                        if a in MY_ARMS]
        c0a = {}
        for a, lin in pairs:
            for s in (801, 901):
                for key in C0A_KEYS:
                    assert f"{key}_{lin}_{s}" in z135c.files, "C0A coverage miss"
                    c0a[f"{a}~{lin}_{key}_{s}"] = float(np.abs(
                        store[f"{key}_{a}_{s}"] - z135c[f"{key}_{lin}_{s}"][:WIN]).max())
                c0a[f"{a}~{lin}_af_{s}"] = abs(float(store[f"af_{a}_{s}"])
                                              - float(z135c[f"af_{lin}_{s}"]))
        mx = max(c0a.values())
        ok = bool(all(v == 0.0 for v in c0a.values()))
        ANCH_STATE.update(c0a135_pass=ok, c0a135_max=mx, c0a135_checks=len(c0a),
                          c0a135_pairs=pairs, g_cutwA=GAIN["cutwA"],
                          g_cutwA_stored_exact=bool(GAIN["cutwA"] == G_CUTWA_STORED))
        atomic_json(ANCH_PATH, ANCH_STATE)
        log(f"C0A-198a {pairs} x 22 keys + af x 2 seeds vs raw_l135: "
            f"{'PASS' if ok else 'FAIL'} (max {mx:.1e}, {len(c0a)} checks)")
        if not ok:
            _void("C0A-198a FAIL")
    if seed == ANCH_175[0]:
        if not ANCH_STATE.get("c0a135_pass"):
            _void("C0A-198a not PASS at the L175 anchor")
        arms175 = ["cutwA"] + [a for a in MY_ARMS if a == "sc25"]
        c0b = {}
        for a in arms175:
            _w = 0 if a == "cutwA" else W175[a]
            zr = np.load(f"l175/raw_w{_w}.npz")
            zx = np.load(f"l175/xm_w{_w}.npz")
            for key in C0A_KEYS:
                c0b[f"{a}_{key}"] = float(np.abs(
                    store[f"{key}_{a}_{seed}"] - zr[f"{key}_{a}_{seed}"]).max())
            c0b[f"{a}_af"] = abs(float(store[f"af_{a}_{seed}"])
                                 - float(zr[f"af_{a}_{seed}"]))
            c0b[f"{a}_xmean"] = float(np.abs(
                XM[f"xmean_{a}_{seed}"] - zx[f"xmean_{a}_{seed}"]).max())
            _mine = store[f"xwin_{a}_{seed}"][:N_TRACK_175]
            _thr = zr[f"xwin_{a}_{seed}"]
            assert _thr.shape == (N_TRACK_175,)
            c0b[f"{a}_xwin"] = (0.0 if np.array_equal(_mine, _thr, equal_nan=True)
                                else 1.0)
        mx = max(c0b.values())
        ok = bool(all(v == 0.0 for v in c0b.values()))
        ANCH_STATE.update(c0a175_pass=ok, c0a175_max=mx, c0a175_checks=len(c0b),
                          c0a175_arms=arms175, c0a175_seed=int(seed))
        atomic_json(ANCH_PATH, ANCH_STATE)
        log(f"C0A-198b {arms175} x (22 keys + af + xmean + xwin) @ {seed} vs "
            f"stored L175 raws: {'PASS' if ok else 'FAIL'} (max {mx:.1e}, "
            f"{len(c0b)} checks)")
        if not ok:
            _void("C0A-198b FAIL")
    if seed == ANCH_187[0]:
        if not (ANCH_STATE.get("c0a135_pass") and ANCH_STATE.get("c0a175_pass")):
            _void("earlier anchors not PASS at the L187 anchor")
        arms187 = ["cutwA"] + [a for a in MY_ARMS if a in A187]
        c0c = {}
        for a in arms187:
            zr = np.load(f"l187/raw_w{W187[a]}.npz")
            zx = np.load(f"l187/xm_w{W187[a]}.npz")
            for key in C0A_KEYS:
                c0c[f"{a}_{key}"] = float(np.abs(
                    store[f"{key}_{a}_{seed}"] - zr[f"{key}_{a}_{seed}"]).max())
            c0c[f"{a}_af"] = abs(float(store[f"af_{a}_{seed}"])
                                 - float(zr[f"af_{a}_{seed}"]))
            c0c[f"{a}_xmean"] = float(np.abs(
                XM[f"xmean_{a}_{seed}"] - zx[f"xmean_{a}_{seed}"]).max())
            _mine = store[f"xwin_{a}_{seed}"][:N_TRACK_187]
            _thr = zr[f"xwin_{a}_{seed}"]
            assert _thr.shape == (N_TRACK_187,)
            c0c[f"{a}_xwin"] = (0.0 if np.array_equal(_mine, _thr, equal_nan=True)
                                else 1.0)
            for _cg in (GI_CARRIER, GI_RELAY, GI_SC40):
                if f"c{_cg}_{a}_{seed}" in zr.files:
                    c0c[f"{a}_c{_cg}"] = float(np.abs(
                        store[f"c{_cg}_{a}_{seed}"] - zr[f"c{_cg}_{a}_{seed}"]).max())
        mx = max(c0c.values())
        ok = bool(all(v == 0.0 for v in c0c.values()))
        ANCH_STATE.update(c0a187_pass=ok, c0a187_max=mx, c0a187_checks=len(c0c),
                          c0a187_arms=arms187, c0a187_seed=int(seed))
        atomic_json(ANCH_PATH, ANCH_STATE)
        log(f"C0A-198c {arms187} x (22 keys + af + xmean + xwin + cell traces) @ "
            f"{seed} vs stored L187 raws: {'PASS' if ok else 'FAIL'} "
            f"(max {mx:.1e}, {len(c0c)} checks)")
        if not ok:
            _void("C0A-198c FAIL")
    if seed in FRESH:
        if not (ANCH_STATE.get("c0a135_pass") and ANCH_STATE.get("c0a175_pass")
                and ANCH_STATE.get("c0a187_pass")):
            _void("fresh seed reached without all three anchors PASS")
