def classify(t):
    if t["meanR"] >= .60 and t["kR"] >= K_SUP:
        return "SUPER"

        if .35 <= t["meanR"] <= .50 and 3 * m / 16 <= t["kR"]:
            return "NOT-SUPER/PARTIAL"
        return "NOT-SUPER"
    return "INTERMEDIATE"

def arm_row(a, z, info):
    L, R = duty_seeds(z, a, SEEDS, "Ti")
    t = dict(kL=int((L >= .5).sum()), kR=int((R >= .5).sum()), n=m,
             meanL=round(float(L.mean()), 3), meanR=round(float(R.mean()), 3))
    lbR, ubR = wilson(t["kR"], m)
    d = R - RwA
    xw = np.mean([z[f"xwin_{a}_{s}"] for s in SEEDS], axis=0)
    row = dict(arm=a, **info, **t, lbR=lbR, ubR=ubR,
               dutyR=[float(v) for v in R], dutyL=[float(v) for v in L],
               dR_vs_wA=round(float(d.mean()), 3),
               n_up_vs_wA=int((d >= .20).sum()), n_dn_vs_wA=int((d <= -.20).sum()),
               gain=gains[a], gain_delta_vs_wA=gains[a] - gains["cutwA"],
               af_mean=round(float(np.mean([float(z[f"af_{a}_{s}"])
                                            for s in SEEDS])), 6),
               x_window_in_arm=[None if np.isnan(v) else round(float(v), 5)
                                for v in xw])
    row["cls"] = classify(t)
    return row

def spearman(xs, ys):
    if len(xs) < 3:
        return None
    rx, ry = pd.Series(xs).rank().to_numpy(), pd.Series(ys).rank().to_numpy()
    if rx.std() == 0 or ry.std() == 0:
        return None
    return float(np.corrcoef(rx, ry)[0, 1])
