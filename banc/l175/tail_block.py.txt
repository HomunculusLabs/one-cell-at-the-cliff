def select(pool_mask, label):
    cand = np.where(pool_mask[gi_wA])[0]            # local idx on the base
    d = drive_loc[cand]
    with np.errstate(divide="ignore", invalid="ignore"):
        score = np.where(d > 0, np.abs(np.log(d / d_car)), np.inf)
        tie = np.where(OW_RAW[gi_wA[cand]] > 0,
                       np.abs(np.log(OW_RAW[gi_wA[cand]] / m_car)), np.inf)
    o = np.lexsort((gi_wA[cand], tie, score))       # score, then tie, then gi
    rows = []
    for j in o[:6]:
        g = int(gi_wA[cand[j]])
        ci = cell_info(g)
        ci.update(x_mean=float(xbar[cand[j]]),
                  x_mean_seeds=[float(v) for v in xm3[:, cand[j]]],
                  delivered_drive=float(d[j]), drive_ratio=float(d[j] / d_car),
                  abs_log_drive_ratio=float(score[j]),
                  mass_ratio=float(OW_RAW[g] / m_car),
                  abs_log_mass_ratio=float(tie[j]))
        rows.append(ci)
    log(f"SELECT {label}: pool {len(cand)} | pick gi {rows[0]['gi']} "
        f"{rows[0]['super_class']}/{rows[0]['cell_type']}/{rows[0]['side']} "
        f"x {rows[0]['x_mean']:.4f} w_eff {rows[0]['out_w_eff']:.2f} "
        f"drive ratio {rows[0]['drive_ratio']:.5f} sign {rows[0]['eff_sign']:+.0f}")
    return dict(pool_n=int(len(cand)), n_positive_drive=int((d > 0).sum()),
                pick=rows[0], runners_up=rows[1:6])
