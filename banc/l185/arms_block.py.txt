def signed_current(xbar, g, gate):
    """PREREG §R3. Rows of Wb are targets, columns are sources. For pool P:
    u_P[m] = (1/|P|) sum_{t in P} Wb[t, m];
    A1_P[c] = u_P[c];  A2_P[c] = sum_{m != c} u_P[m] * gate[m] * Wb[m, c];
    J_P[c] = g * xbar[c] * (A1_P[c] + g * A2_P[c])."""
    n = n_cutk["cutwA"]
    crow = Wb_wA.crow_indices().numpy()
    col = Wb_wA.col_indices().numpy()
    val = Wb_wA.values().numpy().astype(np.float64)
    row = np.repeat(np.arange(n), np.diff(crow))
    assert len(row) == len(col) == cutk_pins["cutwA"]["edges"]
    nd = row != col
    out = {}
    for P, pk in (("f", "Ti_fR"), ("e", "Ti_eR")):
        pool = POOLS["cutwA"][pk]
        inpool = np.zeros(n, bool)
        inpool[pool] = True
        u = np.zeros(n)
        _s = inpool[row]
        np.add.at(u, col[_s], val[_s])
        u /= len(pool)
        A2 = np.zeros(n)
        np.add.at(A2, col[nd], (u * gate)[row[nd]] * val[nd])
        out[P] = dict(A1=u, A2=A2, J1=g * xbar * u, J2=g * xbar * g * A2,
                      J=g * xbar * (u + g * A2))
    return out


def _brute_paths(c_gi, xbar, g, gate):
    """Independent re-derivation of (J_f, J_e) for ONE cell from the global
    edge arrays (float64 log1p weights), not from the CSR matrix."""
    weff = vals_all * np.where(vals_all < 0, BETA, 1.0)
    res = []
    oe = np.where(emask_wA & (src_all == c_gi))[0]
    for pk in ("Ti_fR", "Ti_eR"):
        pg = gi_wA[POOLS["cutwA"][pk]]
        inp = np.zeros(Nall, bool)
        inp[pg] = True
        a1 = float(weff[oe][inp[dst_all[oe]]].sum()) / len(pg)
        a2 = 0.0
        for e in oe:
            m = int(dst_all[e])
            if m == c_gi or gate[loc_wA[m]] == 0:
                continue
            me = emask_wA & (src_all == m) & inp[dst_all]
            a2 += float(weff[me].sum()) / len(pg) * float(weff[e])
        res.append(g * float(xbar[loc_wA[c_gi]]) * (a1 + g * a2))
    return np.array(res)
