def classify(t, n=None):
    n = m if n is None else n
    ks, kn = math.ceil(.75 * n), math.floor(.5 * n)
    if t["meanR"] >= .60 and t["kR"] >= ks:
        return "SUPER"
    if t["kR"] <= kn and t["meanR"] <= .50:
        if .35 <= t["meanR"] <= .50 and 3 * n / 16 <= t["kR"]:
            return "NOT-SUPER/PARTIAL"
        return "NOT-SUPER"
    return "INTERMEDIATE"

def arm_row(a, **info):
    L, R = duty(a)
    z = raws[0] if a == "cutwA" else raws[arm_w[a]]
    t = dict(kL=int((L >= .5).sum()), kR=int((R >= .5).sum()), n=m,
             meanL=round(float(L.mean()), 3), meanR=round(float(R.mean()), 3))
    lbR, ubR = wilson(t["kR"], m)
    d = R - RwA
    row = dict(arm=a, **info, **t, lbR=lbR, ubR=ubR,
               dutyR=[float(v) for v in R], dutyL=[float(v) for v in L],
               dR_vs_wA=round(float(d.mean()), 3),
               n_up_vs_wA=int((d >= .20).sum()), n_dn_vs_wA=int((d <= -.20).sum()),
               gain=gains[a], gain_delta_pct=100 * (gains[a] / G0 - 1),
               gain_source=("lineage stored" if a == "cutwA" else gsrc[a]),
               af_mean=round(float(np.mean([float(z[f"af_{a}_{s}"])
                                            for s in SEEDS])), 6))
    row["cls"] = classify(t)
    return row


def paired(Ra, Rb):
    """a - b on the same seeds."""
    d = np.asarray(Ra) - np.asarray(Rb)
    sd = float(d.std(ddof=1)) if len(d) > 1 else float("nan")
    se = sd / math.sqrt(len(d)) if len(d) > 1 else float("nan")
    return dict(mean=round(float(d.mean()), 4), sd=round(sd, 4),
                t=(round(float(d.mean() / se), 3) if se and se > 0 else None),
                n_up=int((d >= .20).sum()), n_dn=int((d <= -.20).sum()), n=len(d))

def scan_fit(mask):
    if mask.sum() < 3:
        return None
    dd, mm_ = sd_[mask], sm_[mask]
    pred = r175["base"]["meanR"] + slope * (dd * 10)
    ss_tot = float(((mm_ - mm_.mean()) ** 2).sum())
    A = np.vstack([dd * 10, np.ones(len(dd))]).T
    b_scan = float(np.linalg.lstsq(A, mm_, rcond=None)[0][0])
    return dict(n=int(mask.sum()),
                scan_slope_per_0p1pct=round(b_scan, 4),
                r2_of_gain_null_line=round(1 - float(((mm_ - pred) ** 2).sum()) / ss_tot, 3),
                sd_meanR=round(float(mm_.std(ddof=1)), 4),
                slope_ratio_null_over_scan=(round(slope / b_scan, 3) if b_scan else None))
