def _info200(g):
    g = int(g)
    ci = cell_info(g)
    ci.update(x_mean=float(_xb_g[g]), total_drive=float(_drv_g[g]),
              total_drive_ratio=float(_drv_g[g] / D_RU2),
              dL=float(DLg[g]), dR=float(DRg[g]), dL_ratio=float(DLg[g] / DL_REF),
              edges_onto_PM_L=int(N_PM["L"][g]), edges_onto_PM_R=int(N_PM["R"][g]),
              mass_onto_PM_L=float(OW_PM["L"][g]), mass_onto_PM_R=float(OW_PM["R"][g]),
              in_PM_L=bool(is_pm["L"][g]), in_PM_R=bool(is_pm["R"][g]),
              in_edges=int((emask_wA & (dst_all == g)).sum()),
              direct_onto={pk: _direct_onto(g, pk) for pk in _pool_gi},
              edge_from_carrier=_edge(GI_CARRIER, g), edge_to_carrier=_edge(g, GI_CARRIER),
              edge_from_ru2=_edge(GI_RU2, g), edge_to_ru2=_edge(g, GI_RU2))
    return ci


def select_left():
    cand = np.where(POOL_L)[0]                       # global gi
    d = DLg[cand]
    with np.errstate(divide="ignore", invalid="ignore"):
        score = np.where(d > 0, np.abs(np.log(d / DL_REF)), np.inf)
        tie = np.where(_drv_g[cand] > 0, np.abs(np.log(_drv_g[cand] / D_RU2)), np.inf)
    o = np.lexsort((cand, tie, score))               # score, then tie, then gi
    rows = []
    for j in o[:5]:
        r = _info200(cand[j])
        r.update(rank=len(rows) + 1, abs_log_dL_ratio=float(score[j]),
                 abs_log_total_drive_ratio=float(tie[j]))
        rows.append(r)
    return dict(
        rule="ascending |log(dL_c/dL_ru2)| over POOL_L; tie-break |log(total-drive "
             "ratio)|, then gi; dL = x_mean (stored 3-seed census of the k=24 base) x "
             "sum of model-true |Wb| over the cell's base edges onto the left-Ti "
             "premotor pool PM_L (PREREG l200)",
        pool="in k=24 base & cord interneuron & eff sign == ru2's & not prior-arm "
             "(L198 PRIOR + L198's six cells) & not motor & not DNg100 & not the homolog pick",
        census_seeds=[801, 901, 3561], pool_n=int(len(cand)),
        pool_n_positive_dL=int((d > 0).sum()), prior_arm_cells_n=len(PRIOR200),
        PM_L_n=int(len(PM["L"])), PM_R_n=int(len(PM["R"])),
        PM_shared_n=int((is_pm["L"] & is_pm["R"]).sum()),
        dL_ref=DL_REF, ru2=_info200(GI_RU2), top5=rows,
        picks=[int(rows[0]["gi"]), int(rows[1]["gi"])],
        homolog=(_info200(HOM_GI) if HOM_GI is not None else None),
        homolog_candidates=HOM_CAND)

def _a4():
    """A4 mirror-symmetry probe — wiring only (PREREG l200 §A4, metrics fixed there)."""
    em = emask_wA
    rs, ra = {}, {}
    for pk, mn in MN.items():
        mk = em & np.isin(dst_all, mn)
        r = np.zeros(Nall)
        np.add.at(r, src_all[mk], _sgnw_all[mk])
        a = np.zeros(Nall)
        np.add.at(a, src_all[mk], _absw_all[mk])
        rs[pk], ra[pk] = r, a
    nm = em & ~is_motor_g[dst_all]                   # s -> non-motor intermediate p
    A2, S2 = {}, {}
    for pk in MN:
        a = np.zeros(Nall)
        np.add.at(a, src_all[nm], _absw_all[nm] * ra[pk][dst_all[nm]])
        s = np.zeros(Nall)
        np.add.at(s, src_all[nm], _sgnw_all[nm] * rs[pk][dst_all[nm]])
        A2[pk], S2[pk] = a, s
    A_side = {sd: A2[f"Ti_f{sd}"] + A2[f"Ti_e{sd}"] for sd in "LR"}
    with np.errstate(divide="ignore", invalid="ignore"):
        LI2 = (A_side["R"] - A_side["L"]) / (A_side["R"] + A_side["L"])
        LI1 = (OW_PM["R"] - OW_PM["L"]) / (OW_PM["R"] + OW_PM["L"])

    def tgt(s):
        return set(int(x) for x in dst_all[em & (src_all == s)])

    def typed(gs, flip):
        out = set()
        for g in gs:
            if g not in _ann1.index:
                continue
            ct, sd = str(_ann1.loc[g, "cell_type"]), str(_ann1.loc[g, "side"]).lower()
            if ct in ("", "nan") or sd not in ("left", "right"):
                continue
            if flip:
                sd = "left" if sd == "right" else "right"
            out.add((ct, sd))
        return out

    def jac(a, b):
        return (len(a & b) / len(a | b)) if (a | b) else None

    cells = {"carrier": GI_CARRIER, "ru2": GI_RU2, "lt1": LT_GI[0], "lt2": LT_GI[1]}
    if HOM_GI is not None:
        cells["l2h"] = HOM_GI
    out = dict(cells={})
    for name, s in cells.items():
        ci = cell_info(s)
        tot = float(OW_EFF[s])
        out["cells"][name] = dict(
            gi=int(s), cell_type=ci["cell_type"], side=ci["side"], eff_sign=ci["eff_sign"],
            super_class=ci["super_class"], out_edges=int(OW_N[s]), out_mass_eff=tot,
            x_mean=float(_xb_g[s]),
            one_hop=dict(
                PM_L=dict(edges=int(N_PM["L"][s]), mass=float(OW_PM["L"][s]),
                          share=float(OW_PM["L"][s] / tot), drive=float(DLg[s])),
                PM_R=dict(edges=int(N_PM["R"][s]), mass=float(OW_PM["R"][s]),
                          share=float(OW_PM["R"][s] / tot), drive=float(DRg[s])),
                shared=dict(edges=int(N_PMS[s]), mass=float(OW_PMS[s]))),
            LI1=(None if not np.isfinite(LI1[s]) else float(LI1[s])),
            direct_onto={pk: _direct_onto(s, pk) for pk in _pool_gi},
            two_hop={pk: dict(signed=float(S2[pk][s]), abs=float(A2[pk][s])) for pk in MN},
            A_L=float(A_side["L"][s]), A_R=float(A_side["R"][s]),
            LI2=(None if not np.isfinite(LI2[s]) else float(LI2[s])))
    Tc, Tr = tgt(GI_CARRIER), tgt(GI_RU2)
    pmL, pmR = set(int(x) for x in PM["L"]), set(int(x) for x in PM["R"])
    Tc_pm, Tr_pm = Tc & pmR, Tr & pmL
    out["overlap"] = dict(
        n_targets_carrier=len(Tc), n_targets_ru2=len(Tr), n_shared=len(Tc & Tr),
        J_same=jac(Tc, Tr), J_mirror=jac(typed(Tc, False), typed(Tr, True)),
        J_type_sideblind=jac({t for t, _ in typed(Tc, False)}, {t for t, _ in typed(Tr, False)}),
        n_typed_carrier=len(typed(Tc, False)), n_typed_ru2=len(typed(Tr, False)),
        premotor=dict(n_carrier_in_PM_R=len(Tc_pm), n_ru2_in_PM_L=len(Tr_pm),
                      n_carrier_in_PM_L=len(Tc & pmL), n_ru2_in_PM_R=len(Tr & pmR),
                      J_same=jac(Tc_pm, Tr_pm),
                      J_mirror=jac(typed(Tc_pm, False), typed(Tr_pm, True)),
                      mirror_shared_types=sorted(t for t, _ in (typed(Tc_pm, False)
                                                               & typed(Tr_pm, True)))))
    ctx = in_wA & is_cord_g & ((A_side["L"] + A_side["R"]) > 0)
    v = LI2[ctx]
    out["context"] = dict(
        n_cord_with_two_hop=int(ctx.sum()), LI2_median=float(np.median(v)),
        LI2_p10=float(np.percentile(v, 10)), LI2_p90=float(np.percentile(v, 90)),
        ru2_LI2_percentile=float((v <= LI2[GI_RU2]).mean() * 100),
        frac_cord_LI2_le_minus_20=float((v <= -.2).mean()),
        frac_cord_LI2_ge_plus_20=float((v >= .2).mean()))
    lc, lr = out["cells"]["carrier"]["LI2"], out["cells"]["ru2"]["LI2"]
    if lc is not None and lr is not None and lc >= .2 and lr <= -.2:
        lab = "A4_mirror"
    elif (lc is not None and lr is not None and abs(lc) >= .2 and abs(lr) >= .2
          and (lc > 0) == (lr > 0)):
        lab = "A4_same_side"
    else:
        lab = "A4_no_laterality_contrast"
    out["label"] = lab
    out["LI2_carrier"], out["LI2_ru2"] = lc, lr
    return out
