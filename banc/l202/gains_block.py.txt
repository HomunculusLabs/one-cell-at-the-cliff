    log(msg + " -> VOID, stopping before any fresh seed")
    sys.exit(3)


def _replay_cert(seed):
    """CERT-REPLAY: every ru2-replay arm's stored ru2 trace == cutwA's, EXACT."""
    for a in ARMS:
        if a in REPLAY2:
            _d = float(np.abs(store[f"c{GI_RU2}_{a}_{seed}"]
                              - store[f"c{GI_RU2}_cutwA_{seed}"]).max())
            ANCH_STATE["replay_checks"] = int(ANCH_STATE.get("replay_checks", 0)) + 1
            ANCH_STATE["replay_max"] = max(float(ANCH_STATE.get("replay_max", 0.0)), _d)
            if _d != 0.0:
                ANCH_STATE["replay_pass"] = False
                _void(f"CERT-REPLAY FAIL {a} @ {seed} max {_d:.1e}")
    atomic_json(ANCH_PATH, ANCH_STATE)


def post_seed(seed):
    """In-run anchors, all evaluated BEFORE a fresh seed is rolled.
    (a) at 901: cutwA (+ sc25) vs raw_l135 — C0A, lineage form VERBATIM;
        + CERT-NULL (worker 1): bk2 (base graph + ru2 replay line) == cutwA.
    (b..f) at 3545 / 3609 / 3641 / 3673 / 3705: this worker's arms that have
        stored raws in L175 / L187 / L190 / L198 / L200 vs those raws.
    Any failure = VOID before a single fresh seed is rolled."""
    _replay_cert(seed)
    if seed == 901:
        z135c = np.load("l135/raw_l135.npz")
        pairs = [("cutwA", "cutwA")] + [(a, l) for a, l in (("sc25", "cutng25"),)
                                        if a in MY_ARMS]
        c0a = {}
        for a, lin in pairs:
            for s in (801, 901):
                for key in C0A_KEYS:
                    assert f"{key}_{lin}_{s}" in z135c.files, "C0A coverage miss"
                    c0a[f"{a}~{lin}_{key}_{s}"] = float(np.abs(
                        store[f"{key}_{a}_{s}"] - z135c[f"{key}_{lin}_{s}"][:WIN]).max())
                c0a[f"{a}~{lin}_af_{s}"] = abs(float(store[f"af_{a}_{s}"])
                                              - float(z135c[f"af_{lin}_{s}"]))
        mx = max(c0a.values())
        ok = bool(all(v == 0.0 for v in c0a.values()))
        ANCH_STATE.update(c0a135_pass=ok, c0a135_max=mx, c0a135_checks=len(c0a),
                          c0a135_pairs=pairs, g_cutwA=GAIN["cutwA"],
                          g_cutwA_stored_exact=bool(GAIN["cutwA"] == G_CUTWA_STORED))
        atomic_json(ANCH_PATH, ANCH_STATE)
        log(f"C0A-202a {pairs} x 22 keys + af x 2 seeds vs raw_l135: "
            f"{'PASS' if ok else 'FAIL'} (max {mx:.1e}, {len(c0a)} checks)")
        if not ok:
            _void("C0A-202a FAIL")
        if "bk2" in CERT_ARMS:
            cn = {}
            for s in (801, 901):
                for key in C0A_KEYS:
                    cn[f"{key}_{s}"] = float(np.abs(
                        store[f"{key}_bk2_{s}"] - store[f"{key}_cutwA_{s}"]).max())
                cn[f"af_{s}"] = abs(float(store[f"af_bk2_{s}"]) - float(store[f"af_cutwA_{s}"]))
                cn[f"xmean_{s}"] = float(np.abs(
                    XM[f"xmean_bk2_{s}"] - XM[f"xmean_cutwA_{s}"]).max())
            mxn = max(cn.values())
            okn = bool(all(v == 0.0 for v in cn.values()))
            ANCH_STATE.update(null_pass=okn, null_max=mxn, null_checks=len(cn))
            atomic_json(ANCH_PATH, ANCH_STATE)
            log(f"CERT-NULL bk2 (base + ru2 replay) == cutwA @801/901: "
                f"{'PASS' if okn else 'FAIL'} (max {mxn:.1e}, {len(cn)} checks)")
            if not okn:
                _void("CERT-NULL FAIL")
    if seed in ANCH_MAP:
        tag, ntr, amap = ANCH_MAP[seed]
        _prev = [ANCH_MAP[s][0] for s in sorted(ANCH_MAP) if s < seed]
        if not (ANCH_STATE.get("c0a135_pass")
                and all(ANCH_STATE.get(f"{t}_pass") for t in _prev)):
            _void(f"earlier anchors not PASS at the {tag} anchor")
        mine = [a for a in ARMS if a in amap]
        assert mine and mine[0] == "cutwA" and mine == ARMS
        c = {}
        for a in mine:
            _sa, _rd, _sw = amap[a]
            zr = np.load(f"{_rd}/raw_w{_sw}.npz")
            zx = np.load(f"{_rd}/xm_w{_sw}.npz")
            for key in C0A_KEYS:
                c[f"{a}~{_sa}_{key}"] = float(np.abs(
                    store[f"{key}_{a}_{seed}"] - zr[f"{key}_{_sa}_{seed}"]).max())
            c[f"{a}~{_sa}_af"] = abs(float(store[f"af_{a}_{seed}"])
                                     - float(zr[f"af_{_sa}_{seed}"]))
            c[f"{a}~{_sa}_xmean"] = float(np.abs(
                XM[f"xmean_{a}_{seed}"] - zx[f"xmean_{_sa}_{seed}"]).max())
            _thr = zr[f"xwin_{_sa}_{seed}"]
            assert _thr.shape == (ntr,), (tag, a, _thr.shape)
            c[f"{a}~{_sa}_xwin"] = (0.0 if np.array_equal(
                store[f"xwin_{a}_{seed}"][:ntr], _thr, equal_nan=True) else 1.0)
            for _cg in CTR_CELLS:
                if f"c{_cg}_{_sa}_{seed}" in zr.files:
                    assert f"c{_cg}_{a}_{seed}" in store, (tag, a, _cg)
                    c[f"{a}~{_sa}_c{_cg}"] = float(np.abs(
                        store[f"c{_cg}_{a}_{seed}"] - zr[f"c{_cg}_{_sa}_{seed}"]).max())
        mx = max(c.values())
        ok = bool(all(v == 0.0 for v in c.values()))
        ANCH_STATE.update({f"{tag}_pass": ok, f"{tag}_max": mx, f"{tag}_checks": len(c),
                           f"{tag}_arms": [f"{a}~{amap[a][0]}" for a in mine],
                           f"{tag}_seed": int(seed)})
        atomic_json(ANCH_PATH, ANCH_STATE)
        log(f"C0A-202 {tag} {ANCH_STATE[f'{tag}_arms']} x (22 keys + af + xmean + xwin + "
            f"cell traces) @ {seed}: {'PASS' if ok else 'FAIL'} (max {mx:.1e}, "
            f"{len(c)} checks)")
        if not ok:
            _void(f"C0A-202 {tag} FAIL")
    if seed in FRESH:
        if not all(ANCH_STATE.get(k) for k in NEED_ANCH):
            _void("fresh seed reached without all six anchors PASS")
        if WID == 1 and not ANCH_STATE.get("null_pass"):
            _void("fresh seed reached without CERT-NULL PASS")
