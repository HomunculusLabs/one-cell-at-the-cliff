def add_edge_arm(name, src_gi, dst_gi):
    """Base graph minus ONE CSR entry (dst row, src col) — L190 edge_run.py form."""
    add_base_alias(name)
    base = Wb_wA
    row, col = int(loc_wA[dst_gi]), int(loc_wA[src_gi])
    assert row >= 0 and col >= 0
    cr, ci, val = base.crow_indices(), base.col_indices(), base.values()
    lo, hi = int(cr[row]), int(cr[row + 1])
    hits = torch.where(ci[lo:hi] == col)[0]
    assert len(hits) == 1, (name, len(hits))
    j = lo + int(hits[0])
    keep = torch.arange(len(ci)) != j
    newcr = cr.clone()
    newcr[row + 1:] -= 1
    WB[name] = torch.sparse_csr_tensor(newcr, ci[keep], val[keep], size=base.shape)
    w = WB[name]
    assert torch.equal(w.col_indices(), ci[keep]) and torch.equal(w.values(), val[keep])
    assert torch.equal(cr[1:] - cr[:-1] - (w.crow_indices()[1:] - w.crow_indices()[:-1]),
                       (torch.arange(len(cr) - 1) == row).to(cr.dtype))
    assert float(val[j]) != 0.0
    GEN[name]["edges"] -= 1
    assert GEN[name]["edges"] == cutk_pins["cutwA"]["edges"] - 1
    return float(val[j])
